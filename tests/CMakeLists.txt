# nanowire tests. Every test is a plain executable registered with CTest.
function(nanowire_test name)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE nanowire::nanowire nanowire_dev_flags)
  add_test(NAME ${name} COMMAND ${name} ${ARGN})
endfunction()

nanowire_test(test_protocols)
nanowire_test(test_malformed)
nanowire_test(test_reassembly)
nanowire_test(test_dispatch)

# scapy as an independent oracle. Committed vectors (generated by scapy_oracle.py gen) are always
# checked; with a Python that has scapy, fresh vectors are generated and nanowire-built frames are
# verified by scapy too.
add_executable(test_oracle test_oracle.cpp)
target_link_libraries(test_oracle PRIVATE nanowire::nanowire nanowire_dev_flags)
add_test(NAME oracle_scapy_vectors_parse
         COMMAND test_oracle parse ${CMAKE_CURRENT_SOURCE_DIR}/data/scapy_vectors.pcap
                                   ${CMAKE_CURRENT_SOURCE_DIR}/data/scapy_vectors.kv)
set(NANOWIRE_ORACLE_PYTHON "" CACHE FILEPATH "Python interpreter with scapy for the oracle tests (auto-detected)")
if(NOT NANOWIRE_ORACLE_PYTHON)
  foreach(_py ${Python3_EXECUTABLE} /usr/bin/python3 python3)
    execute_process(COMMAND ${_py} -c "import scapy.contrib.automotive.someip, scapy.layers.inet6"
                    RESULT_VARIABLE _rc OUTPUT_QUIET ERROR_QUIET)
    if(_rc EQUAL 0)
      set(NANOWIRE_ORACLE_PYTHON ${_py})
      break()
    endif()
  endforeach()
endif()
if(NANOWIRE_ORACLE_PYTHON)
  message(STATUS "nanowire: scapy oracle tests enabled (${NANOWIRE_ORACLE_PYTHON})")
  set(_o ${CMAKE_CURRENT_SOURCE_DIR}/oracle/scapy_oracle.py)
  set(_b ${CMAKE_CURRENT_BINARY_DIR})
  add_test(NAME oracle_scapy_gen COMMAND ${NANOWIRE_ORACLE_PYTHON} ${_o} gen ${_b}/scapy_fresh.pcap ${_b}/scapy_fresh.kv)
  add_test(NAME oracle_scapy_parse COMMAND test_oracle parse ${_b}/scapy_fresh.pcap ${_b}/scapy_fresh.kv)
  add_test(NAME oracle_nanowire_build COMMAND test_oracle build ${_b}/nanowire_built.pcap ${_b}/nanowire_built.kv)
  add_test(NAME oracle_scapy_check_build
           COMMAND ${NANOWIRE_ORACLE_PYTHON} ${_o} check ${_b}/nanowire_built.pcap ${_b}/nanowire_built.kv)
  set_tests_properties(oracle_scapy_parse PROPERTIES DEPENDS oracle_scapy_gen)
  set_tests_properties(oracle_scapy_check_build PROPERTIES DEPENDS oracle_nanowire_build)
else()
  message(STATUS "nanowire: scapy not found; oracle tests use the committed vectors only")
endif()
